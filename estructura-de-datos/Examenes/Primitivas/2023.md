3. Implementar el constructor de un árbol binario de búsqueda ABB<T> a partir de un vector de STL con datos ordenados de menor a mayor de forma que quede equilibrado en altura (compatible con AVL). Se puede hacer de forma recursiva haciendo que el dato central del vector (completo o parte de éste) sea la raíz

```cpp
ABB<T>:: ABB(const vector<T> &v);
```

```cpp
template <typename T>
class ABB{
    private:
        class Nodo{
            T dato;
            Nodo *izq;
            Nodo *der;
            Nodo(const T &d): dato(d), izq(nullptr),der(nullptr){}
        };
        Nodo *raiz;

        Nodo *construirEquilibrado(const vector<T> &v, int inicio, int fin){
            if(inicio>fin){
                return nullptr;
            }
            int medio=inicio + (fin-inicio)/2;
            Nodo *nuevo = new Nodo(v[medio]);

            nuevo->izq=construirEquilibrado(v, inicio,medio-1);

            nuevo->der=construirEquilibrado(v,medio+1,fin);

            return nuevo;
        }
    public:
        ABB(const vector<T> &v){
            if(v.empty()){
                raiz=nullptr;
            }else{
                raiz=construirEquilibrado(v,0,v.tam()-1);
            }
        }
};
```
